function renderAuditReport(parsed) { if (Array.isArray(parsed)) { let combined = { tabel_audit: [], summary_sertifikat: [] }; parsed.forEach(item => { if (item.tabel_audit) combined.tabel_audit = combined.tabel_audit.concat(item.tabel_audit); if (item.summary_sertifikat) combined.summary_sertifikat = combined.summary_sertifikat.concat(item.summary_sertifikat); }); parsed = combined; } if (!parsed || !parsed.tabel_audit) { console.error("Data tabel_audit tidak ditemukan pada objek:", parsed); return; } let certRows = [], penRows = []; parsed.tabel_audit.forEach(r => { let k = String(r.kategori || "").toUpperCase(); if (k.includes("SERTIFIKAT") || k.includes("HEADER")) certRows.push(r); else penRows.push(r); }); // 1. Populate Top Bar (Data Kapal) let v_name="-", v_imo="-", v_call="-", v_own="-"; for(let r of certRows) { if(String(r.di_sertifikat).toLowerCase().includes("name:")) { String(r.di_sertifikat).split('|').forEach(x => { let kv = x.split(':'); if(kv.length<2) return; let k = kv[0].trim().toLowerCase(), v = kv[1].trim(); if(k.includes("name")) v_name=v; if(k.includes("imo")) v_imo=v; if(k.includes("call")) v_call=v; if(k.includes("owner")) v_own=v; }); break; } } document.getElementById("val-vessel").innerText = v_name.toUpperCase(); document.getElementById("val-imo").innerText = v_imo; document.getElementById("val-call").innerText = v_call; document.getElementById("val-owner").innerText = v_own; if(v_name !== "-") currentShipName = "Audit_" + v_name.replace(/\s+/g, '_'); // PARSING DETEKSI RN DARI SERTIFIKAT LIFERAFT SECARA SPESIFIK (OK DIABAIKAN TOTAL) let certRnItems = []; certRows.forEach(r => { let detailTeks = String(r.detail_logistik_internal?.catatan_komponen_tabel || r.hasil_ai || "").toUpperCase(); if (detailTeks.includes("FIRST AID") || detailTeks.includes("P3K")) { if (detailTeks.includes("RN") || detailTeks.includes("RENEW")) certRnItems.push("First Aid Kit"); } if (detailTeks.includes("SEASICKNESS") || detailTeks.includes("ANTIMO")) { if (detailTeks.includes("RN") || detailTeks.includes("RENEW")) certRnItems.push("Anti-Seasickness"); } }); if(certRnItems.length === 0) { penRows.forEach(r => { let itemStr = String(r.keterangan).toUpperCase(); let certQtyStr = String(r.di_sertifikat).toUpperCase(); if (certQtyStr.includes("RN") || certQtyStr.includes("RENEW")) { if (itemStr.includes("FIRST AID")) certRnItems.push("First Aid Kit"); if (itemStr.includes("SICKNESS") || itemStr.includes("ANTIMO")) certRnItems.push("Anti-Seasickness"); } }); } // 2. Render Tabel Penawaran & Tangkap Anomali let htmlPen = ""; let anomalyList = []; penRows.forEach((r, idx) => { let rawAlias = String(r.alias_sistem || "-").trim(); if (rawAlias.includes("Isi ALIAS_SISTEM") || rawAlias.includes("WAJIB SALIN") || rawAlias.includes("JSON")) { rawAlias = "-"; } let aliasParts = rawAlias.split(" - "); let itemCode = aliasParts[0] || "-"; let systemName = aliasParts.slice(1).join(" - ") || "-"; let hasilStr = String(r.hasil_ai || ""); let isWarn = hasilStr.includes('⚠️') || hasilStr.toLowerCase().includes('anomali') || hasilStr.toLowerCase().includes('beda'); let rowClass = isWarn ? 'row-anomali' : 'row-match'; if (isWarn) { let cleanReason = hasilStr.replace(/✅ MATCH|⚠️ ANOMALI/gi, '').trim(); if(cleanReason.startsWith(':') || cleanReason.startsWith('.')) cleanReason = cleanReason.substring(1).trim(); anomalyList.push(`${esc(r.keterangan)}: ${esc(cleanReason)}`); } let certQty = String(r.di_sertifikat || "-").trim(); if (certQty.toLowerCase().includes("qty aktual") || certQty.length > 30) { certQty = "0"; } let badgeHtml = esc(certQty); if (certQty.toUpperCase().includes("(RN)")) { badgeHtml = `${esc(certQty.replace(/\(RN\)/gi, ''))} RN`; } else if (certQty.toUpperCase().includes("(OK)")) { badgeHtml = `${esc(certQty.replace(/\(OK\)/gi, ''))} OK`; } htmlPen += ` ${idx + 1}
${esc(r.keterangan)}
${esc(systemName)}
${esc(itemCode)}
${esc(r.di_penawaran)}
${badgeHtml}
${esc(r.di_do)}
${isWarn?'⚠️ ANOMALI':'✅ MATCH'} `; if (hasilStr.trim() !== "") { let cleanReason = hasilStr.replace(/✅ MATCH[:]?|⚠️ ANOMALI[:]?/gi, '').trim(); if(cleanReason.startsWith(':') || cleanReason.startsWith('.')) cleanReason = cleanReason.substring(1).trim(); if(cleanReason !== "") { htmlPen += `
${isWarn ? '⚠️ AI Evaluation:' : '✅ AI Evaluation:'} ${esc(cleanReason)}
`; } } }); document.getElementById("body-penawaran").innerHTML = htmlPen; // 3. Render TAB SUMMARY (Peta Status Komponen per Kategori Alat) let criticalContainer = document.getElementById("critical-summary-container"); let summaryHtml = ""; if (anomalyList.length > 0) { let listHtml = anomalyList.map(issue => `
  • ${issue}
    • `).join(''); summaryHtml += `

    ⚠️ Buyer Attention Required (Anomali Ditemukan)

    Daftar masalah ketidakpatuhan fisik / komersial berdasarkan ketentuan regulasi maritim & SOP LSA:

      ${listHtml}
    `; } else { summaryHtml += `

    ✅ Audit Completed: All Core Parameters Match

    Semua kesesuaian fisik, rumus material internal liferaft, dan validitas penggantian komponen (RN) dinyatakan patuh terhadap SOP.

    `; } if (parsed.summary_sertifikat && parsed.summary_sertifikat.length > 0) { summaryHtml += `

    📑 Peta Status Material & Validitas Sertifikasi (11 Sistem Utama)

    `; parsed.summary_sertifikat.forEach((item, idx) => { let noCert = String(item.no_sertifikat || "-").trim(); if (noCert.toLowerCase().includes("tidak ditemukan") || noCert.length > 40) noCert = "-"; let isFound = noCert !== "-"; let valClass = isFound ? "font-weight: 700; color: var(--primary);" : "font-weight: 500; color: var(--text-muted); font-style: italic;"; let componentStatusHtml = `Menunggu Integrasi Berkas Berlangsung`; if (String(item.sistem).toUpperCase().includes("LIFERAFT")) { if (isFound) { if (certRnItems.length > 0) { componentStatusHtml = `
    [RN]:`; certRnItems.forEach(it => { componentStatusHtml += `🛠️ ${esc(it)}`; }); componentStatusHtml += `
    `; } else { // KETIKA OK, TIDAK MENAMPILKAN DETAIL BARANG UTUH LAGI, CUKUP BANNER BERSIH INI: componentStatusHtml = `✅ UTUH (TIDAK ADA PERGANTIAN MATERIAL RN)`; } } else { componentStatusHtml = `⚠️ Sertifikat Tidak Terlampir`; } } else if (String(item.sistem).toUpperCase().includes("FIRE EXTINGUISHER")) { componentStatusHtml = `Karakter: Cek Berat Tabung & Penjumlahan Label (SOP-02)`; } else if (String(item.sistem).toUpperCase().includes("FIXED CO2")) { componentStatusHtml = `Karakter: Deteksi Pilot Cylinder (SOP-03)`; } else if (String(item.sistem).toUpperCase().includes("SCBA") || String(item.sistem).toUpperCase().includes("EEBD")) { componentStatusHtml = `Karakter: Kondisi Kelengkapan Mask/Valve Set (SOP-04)`; } else { componentStatusHtml = `Menunggu Mapping Dokumen`; } summaryHtml += ` `; }); summaryHtml += `
    NO KATEGORI SISTEM KESELAMATAN NOMOR SERTIFIKAT AKTUAL INTERNAL MATERIAL STATUS (RENEWAL REQUIRED)
    ${idx + 1} ${esc(item.sistem)}
    ${esc(noCert)}
    ${componentStatusHtml}
    `; } criticalContainer.innerHTML = summaryHtml; // 4. Render Tabel Sertifikat let htmlCert = ""; let certGroups = {}; certRows.forEach(r => { let cNo = "-"; String(r.di_sertifikat).split('|').forEach(p => { if(p.includes("CertNo:")) cNo = p.split(':')[1].trim(); }); if(!certGroups[cNo]) certGroups[cNo] = []; certGroups[cNo].push(r); }); let gIdx = 1; for (let cNo in certGroups) { let groupWarn = certGroups[cNo].some(r => { let h = String(r.hasil_ai).toLowerCase(); return h.includes('⚠️') || h.includes('anomali') || h.includes('beda'); }); htmlCert += `🛡️ NOMOR SERTIFIKAT: ${esc(cNo)}`; certGroups[cNo].forEach(r => { let d = { cert: "-", ship: "-", owner: "-", imo: "-", cls: "-", gt: "-", nat: "-", call: "-", date: "-", sign: "-" }; String(r.di_sertifikat).split('|').forEach(p => { let kv = p.split(':'); if(kv.length < 2) return; let k = kv[0].trim().toLowerCase(); let v = kv[1].trim(); if(k.includes("name")) d.ship=v; if(k.includes("owner")) d.owner=v; if(k.includes("imo")) d.imo=v; if(k.includes("class")) d.cls=v; if(k.includes("gt")) d.gt=v; if(k.includes("nat")) d.nat=v; if(k.includes("call")) d.call=v; if(k.includes("date")) d.date=v; if(k.includes("sign")) d.sign=v; }); let isWarn = String(r.hasil_ai).includes('⚠️') || String(r.hasil_ai).toLowerCase().includes('beda'); let rowClass = isWarn ? 'row-anomali' : 'row-match'; htmlCert += ` ${gIdx++} ${esc(r.keterangan)} ${esc(cNo)} ${esc(d.ship)} ${esc(d.owner)} ${esc(d.imo)} ${esc(d.cls)} ${esc(d.gt)} ${esc(d.nat)} ${esc(d.call)} ${esc(d.date)} ${esc(d.sign)} ${isWarn?'⚠️ BEDA':'✅ OK'} `; }); } document.getElementById("body-sertifikat").innerHTML = htmlCert; }
